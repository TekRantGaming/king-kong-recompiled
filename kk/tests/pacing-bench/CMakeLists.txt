cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(kk_pacing_bench_tests LANGUAGES CXX)
  enable_testing()
endif()
add_executable(kk-pacing-bench-tests main.cpp)
target_compile_features(kk-pacing-bench-tests PRIVATE cxx_std_23)
target_include_directories(kk-pacing-bench-tests PRIVATE ../../src)
add_test(NAME pacing-benchmark COMMAND kk-pacing-bench-tests)
add_executable(kk-benchmark-route-tests route.cpp)
target_compile_features(kk-benchmark-route-tests PRIVATE cxx_std_23)
target_include_directories(kk-benchmark-route-tests PRIVATE ../../src)
add_test(NAME benchmark-route COMMAND kk-benchmark-route-tests)
if(WIN32)
  add_executable(kk-benchmark-writer-tests writer.cpp)
  target_compile_features(kk-benchmark-writer-tests PRIVATE cxx_std_23)
  target_include_directories(kk-benchmark-writer-tests PRIVATE ../../src)
  add_test(NAME benchmark-writer COMMAND kk-benchmark-writer-tests)
  add_test(NAME game-slot-lock COMMAND powershell -NoProfile -ExecutionPolicy Bypass
    -File "${CMAKE_CURRENT_SOURCE_DIR}/../../../codex/game-slot-tests.ps1"
    -OutputRoot "${CMAKE_CURRENT_BINARY_DIR}")
  set_tests_properties(game-slot-lock PROPERTIES TIMEOUT 90)
endif()
if(WIN32 AND TARGET rex::runtime)
  add_executable(kk-vsync-tests vsync.cpp ../../src/settings.cpp)
  target_include_directories(kk-vsync-tests PRIVATE ../../src)
  target_link_libraries(kk-vsync-tests PRIVATE rex::runtime)
  target_compile_features(kk-vsync-tests PRIVATE cxx_std_23)
  target_compile_definitions(kk-vsync-tests PRIVATE KK_VERSION="${KK_VERSION}")
  rexglue_apply_target_settings(kk-vsync-tests)
  add_custom_command(TARGET kk-vsync-tests POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:rex::runtime> $<TARGET_FILE_DIR:kk-vsync-tests>)
  add_test(NAME vsync-settings COMMAND kk-vsync-tests)
endif()
