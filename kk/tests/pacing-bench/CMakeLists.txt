cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(kk_pacing_bench_tests LANGUAGES CXX)
  enable_testing()
endif()
add_executable(kk-pacing-bench-tests main.cpp)
target_compile_features(kk-pacing-bench-tests PRIVATE cxx_std_23)
target_include_directories(kk-pacing-bench-tests PRIVATE ../../src)
add_test(NAME pacing-benchmark COMMAND kk-pacing-bench-tests)
if(WIN32 AND TARGET rex::runtime)
  add_executable(kk-vsync-tests vsync.cpp ../../src/settings.cpp)
  target_include_directories(kk-vsync-tests PRIVATE ../../src)
  target_link_libraries(kk-vsync-tests PRIVATE rex::runtime)
  target_compile_features(kk-vsync-tests PRIVATE cxx_std_23)
  target_compile_definitions(kk-vsync-tests PRIVATE KK_VERSION="${KK_VERSION}")
  rexglue_apply_target_settings(kk-vsync-tests)
  add_custom_command(TARGET kk-vsync-tests POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:rex::runtime> $<TARGET_FILE_DIR:kk-vsync-tests>)
  add_test(NAME vsync-settings COMMAND kk-vsync-tests)
endif()
